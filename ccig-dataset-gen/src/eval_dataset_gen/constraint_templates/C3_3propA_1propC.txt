:- object(X), hasProperty(X, P1', V1'), hasProperty(X, P2', V2'), hasProperty(X, P3', V3'), not hasProperty(X, P4', V4'), P1'!=P2', P1'!=P3', P2'!=P3', P1'!=P4', P2'!=P4', P3'!=P4'.
